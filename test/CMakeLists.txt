# Compile every public header in isolation, including its public usage requirements.
file(GLOB_RECURSE apiHeaders CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/public/*.h")
set(headerChecks)
foreach(header IN LISTS apiHeaders)
    file(RELATIVE_PATH includePath "${PROJECT_SOURCE_DIR}/public" "${header}")
    get_filename_component(typeName "${header}" NAME_WE)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${typeName}.cpp")
    set(contents "#include \"${includePath}\"\n")
    if(typeName MATCHES "^I[A-Z]")
        string(REPLACE "/" "::" qualifiedType "${includePath}")
        string(REGEX REPLACE "\\.h$" "" qualifiedType "${qualifiedType}")
        string(APPEND contents "#include <type_traits>\n")
        string(APPEND contents "static_assert(std::is_abstract_v<${qualifiedType}>);\n")
        string(APPEND contents "static_assert(std::has_virtual_destructor_v<${qualifiedType}>);\n")
    endif()
    file(WRITE "${source}" "${contents}")
    list(APPEND headerChecks "${source}")
endforeach()

add_library(${PROJECT_NAME}.HeaderCheck OBJECT ${headerChecks})
target_link_libraries(${PROJECT_NAME}.HeaderCheck PRIVATE ${PROJECT_NAME})
UnrealVoxelSimConfigureTarget(${PROJECT_NAME}.HeaderCheck)
